-- ۵. جدول هدر فاکتورها (ارقام به میلیون ریال)
CREATE TABLE invoices (
    invoice_id INT PRIMARY KEY,
    invoice_number VARCHAR(30) NOT NULL,
    invoice_date DATE NOT NULL, -- تاریخ میلادی معادل پاییز ۱۴۰۵ / ۲۰۲۶
    due_date DATE NOT NULL,
    customer_id INT REFERENCES customers(customer_id),
    branch_id INT REFERENCES branches(branch_id),
    sales_rep_id INT REFERENCES users(user_id),
    total_amount_m_rial DECIMAL(12, 2) NOT NULL,
    discount_m_rial DECIMAL(12, 2) DEFAULT 0,
    net_amount_m_rial DECIMAL(12, 2) NOT NULL,
    payment_status VARCHAR(30) NOT NULL, -- PAID (تسویه‌شده), CHEQUE_PENDING (چک در جریان), OVERDUE (معوق)
    payment_method VARCHAR(50) NOT NULL
);
